% Rounded panel in a template colour. Arguments: colour, opacity,
% top-left, bottom-right. Coordinates are bp from the top-left.
\newcommand{\fspanel}[4]{%
  \begin{tikzpicture}[remember picture, overlay, x=1bp, y=-1bp,
      shift={(current page.north west)}]
    \path[fill=#1, fill opacity=#2, rounded corners=12bp] (#3) rectangle (#4);
  \end{tikzpicture}}
