% \fsdisk{x centre}{y top}{width}{height}{ellipse radius}{colour}
\newcommand{\fsdisk}[6]{%
  \path[draw, line width=0.6bp, fill=#6]
    (#1-#3/2, #2) -- (#1-#3/2, #2+#4-#5)
    arc[start angle=180, end angle=0, x radius=#3/2, y radius=#5]
    -- (#1+#3/2, #2)
    arc[start angle=0, end angle=-180, x radius=#3/2, y radius=#5]
    -- cycle;
  \draw[line width=0.6bp] (#1, #2) ellipse [x radius=#3/2, y radius=#5];}
